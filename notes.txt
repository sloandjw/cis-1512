**maintenance**

//install Process Manager - manages the Node.js process lifecycle, ensuring uptime and restarts.
npm install -g pm2
pm2 start server.js --name arcadehub
pm2 save
pm2 startup

//install Nginx - acts as a reverse proxy, handling SSL/TLS, load balancing, and routing traffic to your Node.js app
apt install -y nginx
nano /etc/nginx/sites-available/arcadehub

//configure Nginx
server {
    listen 80;
    server_name _;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}

//enable it
ln -s /etc/nginx/sites-available/arcadehub /etc/nginx/sites-enabled/
nginx -t
systemctl restart nginx

//firewall
ufw allow OpenSSH
ufw allow 'Nginx Full'
ufw enable
