//install Nginx - acts as a reverse proxy, handling SSL/TLS, load balancing, and routing traffic to your Node.js app
apt install -y nginx
nano /etc/nginx/sites-available/main

//configure Nginx    

server {
    listen 80 default_server;
    listen [::]:80 default_server;

    server_name _;

    root /var/www/html;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }

    location /cis-1440/ {
        alias /var/www/cis-1440/;
        index index.html;
        try_files $uri $uri/ $uri/index.html =404;
    }

    location /cis-1512/ {
        alias /var/www/cis-1512/;
        index index.html;
        try_files $uri $uri/ $uri/index.html =404;
    }

    location /cis-1512/api/ {
        proxy_pass http://127.0.0.1:3000/api/;
    }

    location /cis-1512/leaderboard {
        proxy_pass http://127.0.0.1:3000/leaderboard;
    }
    
    location /cis-1512/debug {
        proxy_pass http://127.0.0.1:3000/debug;
    }
}
