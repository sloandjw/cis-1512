**maintenance**

//install Process Manager - manages the Node.js process lifecycle, ensuring uptime and restarts.
npm install -g pm2
pm2 start server.js --name arcadehub
pm2 save
pm2 startup

//install Nginx - acts as a reverse proxy, handling SSL/TLS, load balancing, and routing traffic to your Node.js app
apt install -y nginx
nano /etc/nginx/sites-available/arcadehub

//configure Nginx                                              
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    server_name _;

    root /root/cis-1512;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }

    location /leaderboard {
        proxy_pass http://127.0.0.1:3000/leaderboard;
    }

    location /api/ {
        proxy_pass http://127.0.0.1:3000;
    }
}

//may need to change permissions (not sure if this was necessary)
sudo chmod o+rx /root
sudo chmod o+rx /root/cis-1512
sudo chmod -R o+rX /root/cis-1512/arcadehub-server

//enable it
ln -s /etc/nginx/sites-available/arcadehub /etc/nginx/sites-enabled/
nginx -t
systemctl restart nginx

//firewall
ufw allow OpenSSH
ufw allow 'Nginx Full'
ufw enable
